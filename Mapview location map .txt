    // =========================================================

    private BaflocatorMap? BaflocatorMap;


    // =========================================================
    // MAP CENTER
    // =========================================================

        = 5.473921;

    private double CenterLongitude { get; set; }
        = 10.417832;


    // =========================================================
    // SEARCH
    // =========================================================

    private string SearchText { get; set; }

    private List<MapPlace> SearchResults { get; set; }
        = [];


    // =========================================================
    // PLACES
    // =========================================================

    private List<MapPlace> Places { get; set; }
        = [];


    // =========================================================
    // INITIALISATION
    // =========================================================

    protected override async Task OnInitializedAsync()

    }

                District =

                Latitude =
                    5.476200,

                Longitude =
                    10.421000
            }
        ];
    }


    // =========================================================
    // SEARCH
    // =========================================================

    private void OnSearchChanged(
        ChangeEventArgs args)
    {
        SearchText =
            args.Value?.ToString()
            ?? string.Empty;


        if (string.IsNullOrWhiteSpace(
            SearchText))
        {
            SearchResults.Clear();

            return;
        }


        var search =
            SearchText.Trim();


        SearchResults =
            Places
                .Where(x =>
                    x.Name.Contains(
                        search,
                        StringComparison
                            .OrdinalIgnoreCase)

                    ||

                    x.Reference.Contains(
                        search,
                        StringComparison
                            .OrdinalIgnoreCase)

                    ||

                    x.District.Contains(
                        search,
                        StringComparison
                            .OrdinalIgnoreCase))
                .Take(5)
                .ToList();
    }


    private void ClearSearch()
    {
        SearchText =
            string.Empty;

        SearchResults.Clear();
    }


    // =========================================================
    // SEARCH RESULT
    // =========================================================

    private async Task SelectSearchResult(
        MapPlace place)
    {
        SearchText =
            place.Name;

        SearchResults.Clear();

        CenterLatitude =
            place.Latitude;

        CenterLongitude =
            place.Longitude;


        if (BaflocatorMap is not null)
        {
            await BaflocatorMap.CenterOn(
                place.Latitude,
                place.Longitude,
                18);
        }
    }


    // =========================================================
    // MAP PLACE SELECTED
    // =========================================================

    private void OnPlaceSelected(
        MapPlace place)
    {
        NavigationManager.NavigateTo(
            $"/adresse/{place.Reference}");
    }


    // =========================================================
    // USER LOCATION
    // =========================================================

    private async Task LocateUser()
    {
        if (BaflocatorMap is null)
            return;

        await BaflocatorMap.LocateUser();
    }


    private Task OnUserLocationReceived(
        BaflocatorMap.LocationPosition position)
    {
        CenterLatitude =
            position.Latitude;

        CenterLongitude =
            position.Longitude;

        return InvokeAsync(
            StateHasChanged);
    }


    // =========================================================
    // LISTE
    // =========================================================

    private void ShowList()
    {
        NavigationManager.NavigateTo(
            "/adresses");
    }


    // =========================================================
    // BACK
    // =========================================================

    private void GoBack()
    {
        NavigationManager.NavigateTo("/");
    }
}
