window.baflocatorRequests = {

    initialize: function () {

        const cards =
            document.querySelectorAll(
                ".request-card"
            );

        cards.forEach(card => {

            card.addEventListener(
                "keydown",
                function (event) {

                    if (
                        event.key === "Enter" ||
                        event.key === " "
                    ) {

                        event.preventDefault();

                        card.click();
                    }

                }
            );

        });

    }

};
