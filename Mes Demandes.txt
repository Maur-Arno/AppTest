@code {

    // =========================================================
    // STATE
    // =========================================================

    private List<AddressRequest> Requests { get; set; }
        = new();


    // =========================================================
    // INITIALIZATION
    // =========================================================

    protected override async Task OnInitializedAsync()
    {
        /*
         * Dans la version réelle :
         *
         * Requests =
         *     await RequestService
         *         .GetMyRequestsAsync();
         *
         * Pour le moment, données de démonstration.
         */

        Requests = new List<AddressRequest>
        {
            new AddressRequest
            {
                Id = "REQ-00125",
                PlaceName = "Restaurant La Paix",
                Reference = "BAF-DJE-C-000125",
                Status = RequestStatus.Pending
            },

            new AddressRequest
            {
                Id = "REQ-00126",
                PlaceName = "Boutique Mboa",
                Reference = "BAF-KAM-C-000126",
                Status = RequestStatus.Validated
            }
        };

        await Task.CompletedTask;
    }


    // =========================================================
    // NAVIGATION
    // =========================================================

    private void GoBack()
    {
        NavigationManager.NavigateTo("/");
    }


    private void AddAddress()
    {
        NavigationManager.NavigateTo(
            "/ajouter-adresse");
    }


    private void OpenRequest(AddressRequest request)
    {
        if (string.IsNullOrWhiteSpace(request.Id))
            return;

        NavigationManager.NavigateTo(
            $"/mes-demandes/{request.Id}");
    }


    // =========================================================
    // STATUS
    // =========================================================

    private static string GetStatusLabel(
        RequestStatus status)
    {
        return status switch
        {
            RequestStatus.Pending =>
                "En attente",

            RequestStatus.Validated =>
                "Validée",

            RequestStatus.Rejected =>
                "Rejetée",

            RequestStatus.Merged =>
                "Fusionnée",

            _ =>
                "Inconnue"
        };
    }


    private static string GetStatusCssClass(
        RequestStatus status)
    {
        return status switch
        {
            RequestStatus.Pending =>
                "status-pending",

            RequestStatus.Validated =>
                "status-validated",

            RequestStatus.Rejected =>
                "status-rejected",

            RequestStatus.Merged =>
                "status-merged",

            _ =>
                string.Empty
        };
    }


    // =========================================================
    // MODEL
    // =========================================================

    private sealed class AddressRequest
    {
        public string Id { get; set; }
            = string.Empty;

        public string PlaceName { get; set; }
            = string.Empty;

        public string Reference { get; set; }
            = string.Empty;

        public RequestStatus Status { get; set; }
    }


    private enum RequestStatus
    {
        Pending,
        Validated,
        Rejected,
        Merged
    }
}
