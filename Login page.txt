@code {

    // =========================================================
    // MODEL
    // =========================================================

    private LoginModel LoginModel { get; set; }
        = new();


    // =========================================================
    // STATE
    // =========================================================

    private bool IsLoading { get; set; }

    private bool PasswordVisible { get; set; }

    private string? ErrorMessage { get; set; }


    private string PasswordInputType =>
        PasswordVisible
            ? "text"
            : "password";


    // =========================================================
    // LOGIN
    // =========================================================

    private async Task HandleLogin()
    {
        ErrorMessage = null;

        IsLoading = true;

        try
        {
            /*
             * Ici tu appelleras ton service :
             *
             * var result =
             *     await AuthService.LoginAsync(
             *         LoginModel.Identifier,
             *         LoginModel.Password);
             *
             * if (!result.Success)
             * {
             *     ErrorMessage =
             *         result.ErrorMessage;
             *
             *     return;
             * }
             *
             * NavigationManager.NavigateTo(
             *     "/");
             */


            // Simulation temporaire
            await Task.Delay(800);

            if (LoginModel.Identifier
                    != "admin"
                ||
                LoginModel.Password
                    != "123456")
            {
                ErrorMessage =
                    "Identifiant ou mot de passe incorrect.";

                return;
            }


            NavigationManager.NavigateTo("/");
        }
        catch
        {
            ErrorMessage =
                "Une erreur est survenue. Veuillez réessayer.";
        }
        finally
        {
            IsLoading = false;
        }
    }


    // =========================================================
    // PASSWORD
    // =========================================================

    private async Task TogglePassword()
    {
        PasswordVisible =
            !PasswordVisible;

        await JS.InvokeVoidAsync(
            "baflocatorLogin.focusPassword");
    }


    // =========================================================
    // FORGOT PASSWORD
    // =========================================================

    private void ForgotPassword()
    {
        NavigationManager.NavigateTo(
            "/mot-de-passe-oublie");
    }


    // =========================================================
    // CREATE ACCOUNT
    // =========================================================

    private void CreateAccount()
    {
        NavigationManager.NavigateTo(
            "/inscription");
    }


    // =========================================================
    // MODEL
    // =========================================================

    public sealed class LoginModel
    {
        [Required(
            ErrorMessage =
                "Le téléphone ou l'email est obligatoire.")]
        public string Identifier { get; set; }
            = string.Empty;


        [Required(
            ErrorMessage =
                "Le mot de passe est obligatoire.")]
        [MinLength(
            6,
            ErrorMessage =
                "Le mot de passe doit contenir au moins 6 caractères.")]
        public string Password { get; set; }
            = string.Empty;
    }
}
